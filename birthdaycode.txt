        let brainSpeechTimeout = null;
        let brainSpeaking = false;

        function brainSpeak(text, expression, callback) {
            if (brainSpeaking) {
                brainSpeaking = false;
                clearTimeout(brainSpeechTimeout);
            }

            const bubble = document.getElementById('brain-speech-bubble');
            const f = document.getElementById('face');
            if (!bubble || !f) return;

            f.classList.remove('talking', 'is-sad', 'is-happy', 'is-surprised', 'is-normal');
            if (expression === 'happy') {
                f.classList.add('is-happy');
            } else if (expression === 'sad') {
                f.classList.add('is-sad');
            } else if (expression === 'surprised') {
                f.classList.add('is-surprised');
            } else {
                f.classList.add('is-normal');
            }

            bubble.innerHTML = "";
            f.classList.add('talking');
            brainSpeaking = true;

            let i = 0;
            function type() {
                if (!brainSpeaking) return;
                if (i < text.length) {
                    bubble.innerHTML += text.charAt(i);
                    i++;
                    brainSpeechTimeout = setTimeout(type, 30);
                } else {
                    brainSpeaking = false;
                    f.classList.remove('talking');
                    if (typeof callback === 'function') {
                        callback();
                    }
                }
            }
            type();
        }

        function drawNeuralConnections() {
            const container = document.querySelector('.neural-container');
            if (!container) return;

            const rect = container.getBoundingClientRect();

            const n1 = document.getElementById('node-memories');
            const n2 = document.getElementById('node-vault');
            const n3 = document.getElementById('node-future');

            if (!n1 || !n2 || !n3) return;

            function getCenter(el) {
                const r = el.getBoundingClientRect();
                return {
                    x: r.left - rect.left + r.width / 2,
                    y: r.top - rect.top + r.height / 2
                };
            }

            const p1 = getCenter(n1);
            const p2 = getCenter(n2);
            const p3 = getCenter(n3);

            // Curve path 1: memories (left) to vault (bottom-center)
            const cp1 = { x: (p1.x + p2.x) / 2 - 30, y: (p1.y + p2.y) / 2 - 10 };
            document.getElementById('path-memories-vault').setAttribute('d', `M ${p1.x} ${p1.y} Q ${cp1.x} ${cp1.y} ${p2.x} ${p2.y}`);

            currentQuizNode = node;

            selectedAnswerIdx = null;
            wrongAnswersChosen = [];

        function renderQuizQuestion(shouldSpeak = false) {
            const container = document.getElementById(`quiz-container-${currentQuizNode}`);
            if (!container) return;

            const qData = quizData[currentQuizNode][currentQuestionIdx];
            const totalQ = quizData[currentQuizNode].length;

            const hasImages = qData.answers.some(ans => ans.image);

            let answersHTML = "";
            qData.answers.forEach((ans, idx) => {
                let btnClass = "";
                if (selectedAnswerIdx === idx) {
                    btnClass = "correct";
                } else if (wrongAnswersChosen.includes(idx)) {
                    btnClass = "wrong";
                }

                if (hasImages) {
                    answersHTML += `
                        <button class="quiz-answer-btn image-card ${btnClass}" onclick="selectQuizAnswer(${idx})">
                            <img src="${ans.image}" alt="${ans.text}" class="quiz-card-img" onerror="this.style.display='none'">
                            <span class="quiz-card-text">${ans.text}</span>
                        </button>
                    `;
                } else {
                    answersHTML += `
                        <button class="quiz-answer-btn ${btnClass}" onclick="selectQuizAnswer(${idx})">
                            ${ans.text}
                        </button>
                    `;
                }
            });

            container.innerHTML = `
                <div class="quiz-container">
                    <div class="quiz-header">Question ${currentQuestionIdx + 1} of ${totalQ}</div>
                    <div class="quiz-answers-grid ${hasImages ? 'has-images' : ''}">
                        ${answersHTML}
                    </div>
                    <div id="quiz-action-area" style="min-height: 50px; display: flex; justify-content: flex-end;">
                        ${selectedAnswerIdx !== null ? `
                            <button class="quiz-next-btn" onclick="${currentQuestionIdx === totalQ - 1 ? 'finishQuiz()' : 'nextQuizQuestion()'}">
                                ${currentQuestionIdx === totalQ - 1 ? 'Finish ➔' : 'Next Question ➔'}
                            </button>
                        ` : ''}
                    </div>
                </div>
            `;

            if (shouldSpeak) {
                let expr = "normal";
                if (currentQuizNode === 'memories' && currentQuestionIdx === 1) {
                    expr = "happy";
                }

                const grid = container.querySelector('.quiz-answers-grid');
                if (grid) {
                    grid.classList.add('initial-hide');
                }

                brainSpeak(qData.question, expr, () => {
                    if (grid) {
                        grid.classList.remove('initial-hide');
                    }
                });
            }
        }

        function selectQuizAnswer(idx) {
            const qData = quizData[currentQuizNode][currentQuestionIdx];
            const answer = qData.answers[idx];

            if (answer.isCorrect) {
                selectedAnswerIdx = idx;
                wrongAnswersChosen = []; // instantly clear any red wrong highlights
                // Play audio if the question has a custom audio file configured!
                if (qData.audioFile) {
                    playQuizSong(qData.audioFile);
                }
            } else {
                // Clear any other wrong highlights so only the latest clicked is active
                wrongAnswersChosen = [idx];

                // Cyber transition: automatically remove the red alert color after a brief moment
                setTimeout(() => {
                    const btn = document.querySelectorAll(`.quiz-answers-grid button`)[idx];
                    if (btn && btn.classList.contains('wrong')) {
                        btn.classList.remove('wrong');
                        wrongAnswersChosen = wrongAnswersChosen.filter(item => item !== idx);
                    }
                }, 1200);
            }

            // Re-render to highlight selected/wrong answers
            renderQuizQuestion(false);

            // Speak unique reaction and make dynamic expression
            brainSpeak(answer.reaction, answer.expression);
        }

        function nextQuizQuestion() {
            currentQuestionIdx++;

        // --- BACKGROUND CANVAS PARTICLES ---
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        let dots = [];
        function res() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
        window.addEventListener('resize', res); res();

        class Dot {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.r = Math.random() * 3 + 1;
                this.vx = Math.random() * 0.4 - 0.2;
                this.vy = Math.random() * 0.4 - 0.2;
            }
            u() {
                this.x += this.vx;
                this.y += this.vy;
                if (this.x < 0 || this.x > canvas.width) this.vx *= -1;
                if (this.y < 0 || this.y > canvas.height) this.vy *= -1;
            }
            d() {
                ctx.fillStyle = '#9d50bb';
                ctx.globalAlpha = 0.5;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.r, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        for (let i = 0; i < 80; i++) dots.push(new Dot());

        function frame() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            dots.forEach(d => { d.u(); d.d(); });
            requestAnimationFrame(frame);
        }
